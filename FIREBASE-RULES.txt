  function signed(){return request.auth != null;}

  function missionOpen(uid,id){let m=mission(id);let w=world(m.world);return paid(uid,w) && worldOpen(uid,w) && (m.previous == '' || verified(uid,m.previous));}
  function validProfile(p){return p.keys().hasOnly(['name','path','avatar','business']) && text(p.name,1,32) && text(p.business,0,500) && p.path in ['Creator','Affiliate','Service','Digital products','Coach','Local business'] && p.avatar in ['Compass','Feather','Gem','Flame','Shield','Crown'];}
  function validMission(d){return d.keys().hasOnly(['step','status','draft','proofId','feedback','practiceAnswer','updatedAt']) && d.step is int && d.step >= 1 && d.step <= 6 && d.status in ['active','pending','verified'] && text(d.draft,0,6000) && text(d.proofId,0,128) && text(d.feedback,0,2000) && d.practiceAnswer is int && d.updatedAt == request.time;}
  function playerDecision(uid,id){
   let n=request.resource.data;
   let o=resource.data;
   let proof=getAfter(/databases/$(database)/documents/sparkProofs/$(o.proofId)).data;
   return self(uid) && o.status == 'pending' && o.step == 5
    && n.proofId == o.proofId && n.draft == o.draft && n.practiceAnswer == o.practiceAnswer
    && proof.uid == uid && proof.missionId == id
    && ((n.status == 'active' && n.step == 4 && n.feedback == o.feedback && proof.status == 'withdrawn')
     || (n.status == 'verified' && n.step == 6 && proof.status == 'verified'
      && proof.reviewer == uid && proof.reviewedAt == request.time
      && proof.reviewMethod in ['player-confirmed','player-override'] && n.feedback == proof.feedback));
  }
  function ownStep(uid,id){let n=request.resource.data;let o=resource.data;return self(uid) && missionOpen(uid,id) && o.status == 'active' && n.status == 'active' && n.step == o.step+1 && n.step <= 4 && n.proofId == o.proofId && n.feedback == o.feedback && (o.step == 2 ? text(n.draft,60,6000) : n.draft == o.draft) && (o.step == 1 ? n.practiceAnswer == mission(id).correct : n.practiceAnswer == o.practiceAnswer);}

  function reviewStep(uid,id){let n=request.resource.data;let o=resource.data;let proof=getAfter(/databases/$(database)/documents/sparkProofs/$(o.proofId)).data;return owner() && request.auth.uid != uid && o.status == 'pending' && o.step == 5 && n.proofId == o.proofId && n.draft == o.draft && n.practiceAnswer == o.practiceAnswer && proof.uid == uid && proof.missionId == id && proof.feedback == n.feedback && ((n.status == 'verified' && n.step == 6 && proof.status == 'verified') || (n.status == 'active' && n.step == 4 && proof.status == 'rejected'));}
  match /sparkConfig/owner {allow read: if signed();allow write: if false;}

   allow update: if self(uid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['profile','cosmetics','updatedAt']) && validProfile(request.resource.data.profile) && request.resource.data.cosmetics is map && request.resource.data.cosmetics.keys().hasOnly(['title','aura']) && request.resource.data.updatedAt == request.time;
   match /bank/{id} {

    function capacityChange(operation){let c=getAfter(/databases/$(database)/documents/sparkBankCapacity/global).data;let old=exists(/databases/$(database)/documents/sparkBankCapacity/global)?get(/databases/$(database)/documents/sparkBankCapacity/global).data.used:0;return c.lastSlot == id && c.operation == operation && c.updatedAt == request.time && c.used == old+(operation == 'create'?1:-1) && c.used >= 0 && c.used <= 400;}

   }
   match /missions/{id} {
